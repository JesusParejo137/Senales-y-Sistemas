\chapter{Tema 2: Señales y Sistemas}

\section{Clasificación de las Señales}

\begin{tcolorbox}[cajaResumen, title={Conceptos Fundamentales: Clasificación}]
Una señal es una función matemática dependiente del tiempo $f(t)$ que describe la variación de una variable física que transporta información codificada.
\begin{itemize}[leftmargin=*]
    \item \textbf{Según el medio físico de transmisión}:
    \begin{itemize}
        \item \textit{Eléctricas}: Voltajes o intensidades transmitidos por cables conductores.
        \item \textit{Electromagnéticas}: Ondas en el espacio libre o guías de onda (radio, satélite).
        \item \textit{Ópticas}: Pulsos luminosos guiados por fibra óptica.
    \end{itemize}
    \item \textbf{Según los valores de la función}:
    \begin{itemize}
        \item \textit{Digitales}: Toman un número finito de valores discretos (ej. niveles binarios $0$ y $V$ voltios).
        \item \textit{Analógicas}: Toman un número infinito de valores continuos (ej. señal telefónica).
    \end{itemize}
    \item \textbf{Según los valores del tiempo}:
    \begin{itemize}
        \item \textit{Continuas}: Definidas para todo instante de tiempo $t$.
        \item \textit{Discretas}: Definidas solo en instantes predeterminados de tiempo.
    \end{itemize}
    \item \textbf{Según su periodicidad}:
    \begin{itemize}
        \item \textit{Periódicas}: Cumplen $f(t) = f(t + T)$, donde $T$ es el período fundamental en segundos.
        \item \textit{No periódicas}: La forma de onda no se repite a lo largo del tiempo.
    \end{itemize}
\end{itemize}
\end{tcolorbox}

\subsection{Caso Parámetrico: Señal Senoidal Pura}
Una señal senoidal pura es la señal periódica fundamental y viene dada por:
\begin{equation}
    f(t) = A \sin(\omega t + \phi) = A \sin(2\pi f t + \phi)
\end{equation}
donde $A$ es la amplitud de pico, $T$ es el período, $f = 1/T$ es la frecuencia en Hz, $\omega = 2\pi/T = 2\pi f$ es la frecuencia angular en rad/s, y $\phi$ es el desfase en radianes.

La \textbf{longitud de onda} ($\lambda$) es la distancia espacial que recorre una onda simple en un período $T$ a la velocidad de propagación $c$:
\begin{equation}
    v_{prop} = \frac{\lambda}{T} \implies \lambda = \frac{c}{f} = c \cdot T
\end{equation}
En el vacío, la velocidad de las ondas electromagnéticas es $c = 3 \cdot 10^8$ m/s.

\section{Dominio del Tiempo y de la Frecuencia}

\begin{tcolorbox}[cajaResumen, title={Representación Temporal vs. Frecuencial}]
\begin{itemize}[leftmargin=*]
    \item \textbf{Dominio del tiempo ($t$)}: Muestra la evolución dinámica de la señal según transcurre el tiempo.
    \item \textbf{Dominio de la frecuencia ($\omega$ o $f$)}: Muestra la amplitud y la fase de las componentes senoidales puras (armónicos) que forman la señal. Un armónico puro $f(t) = V \cos(\omega_0 t)$ genera una sola línea en el dominio de la frecuencia.
\end{itemize}
\end{tcolorbox}

\subsection{Repaso de Números Complejos y Fórmulas de Euler}
\begin{itemize}[leftmargin=*]
    \item \textbf{Unidad imaginaria}: $j = \sqrt{-1}$.
    \item \textbf{Formas binómica y polar}: $z = a + j b = |z| e^{j \phi}$, con $|z| = \sqrt{a^2 + b^2}$ y $\phi = \arctan(b/a)$.
    \item \textbf{Fórmulas de Euler}:
    \begin{equation}
        \cos(x) = \frac{e^{jx} + e^{-jx}}{2}, \quad \sin(x) = \frac{e^{jx} - e^{-jx}}{2j}
    \end{equation}
    \item \textbf{Fase según la posición de un número complejo}:
    \begin{itemize}
        \item Real positivo: $\phi = 0$
        \item Real negativo: $\phi = \pi$
        \item Imaginario puro positivo: $\phi = \frac{\pi}{2}$
        \item Imaginario puro negativo: $\phi = -\frac{\pi}{2}$
    \end{itemize}
\end{itemize}

\section{Serie de Fourier para Señales Periódicas}

Toda función $f(t)$ continua y periódica con período $T$ se descompone en una suma infinita de funciones senoidales puras denominadas armónicos.

\subsection{Forma Trigonométrica}
\begin{equation}
    f(t) = \frac{a_0}{T} + \frac{2}{T} \sum_{n=1}^{\infty} \left( a_n \cos(\omega_n t) + b_n \sin(\omega_n t) \right)
\end{equation}
con $\omega_n = n \omega_1 = \frac{2\pi}{T} n$ y coeficientes:
\begin{equation}
    a_n = \int_{-T/2}^{T/2} f(t) \cos(\omega_n t) \, dt, \quad b_n = \int_{-T/2}^{T/2} f(t) \sin(\omega_n t) \, dt
\end{equation}

Forma compacta en cosenos desfasados:
\begin{equation}
    f(t) = \frac{a_0}{T} + \frac{2}{T} \sum_{n=1}^{\infty} |c_n| \cos(\omega_n t + \theta_n)
\end{equation}
donde $c_n = a_n - j b_n = |c_n| e^{j \theta_n}$, $|c_n| = \sqrt{a_n^2 + b_n^2}$ y $\theta_n = \arctan\left(-\frac{b_n}{a_n}\right)$.

\subsection{Forma Exponencial Compleja}
\begin{equation}
    f(t) = \frac{1}{T} \sum_{n=-\infty}^{\infty} c_n e^{j \omega_n t}, \quad c_n = \int_{-T/2}^{T/2} f(t) e^{-j \omega_n t} \, dt
\end{equation}

\subsection{Espectros Discretos y Teorema de Parseval}
\begin{itemize}[leftmargin=*]
    \item \textbf{Espectro de Amplitud}: Gráfica del módulo $|c_n|$ frente a la frecuencia $\omega$.
    \item \textbf{Espectro de Fase}: Gráfica del argumento $\angle c_n = \theta_n$ frente a la frecuencia $\omega$.
    \item \textbf{Teorema de Parseval}: La potencia media disipada por la señal en una resistencia de $1\,\Omega$ equivale a la suma de las potencias de cada armónico individual:
    \begin{equation}
        \bar{P} = \frac{1}{T} \int_{-T/2}^{T/2} |f(t)|^2 \, dt = \sum_{n=-\infty}^{\infty} \left| \frac{c_n}{T} \right|^2
    \end{equation}
\end{itemize}

\section{Tren de Pulsos Rectangulares Periódico}

Para un tren de pulsos rectangulares de amplitud $V$, anchura $\tau$ y período $T$ centrado en $t=0$:
\begin{equation}
    c_0 = a_0 = V\tau, \quad c_n = V\tau \frac{\sin(\omega_n \tau / 2)}{\omega_n \tau / 2} = V\tau \, \text{sinc}\left(\frac{\omega_n \tau}{2}\right), \quad \theta_n = 0
\end{equation}

\begin{tcolorbox}[cajaResumen, title={Propiedades del Tren de Pulsos}]
\begin{itemize}[leftmargin=*]
    \item Su espectro es discreto con infinitos armónicos a frecuencias múltiplos de $\omega_1 = 2\pi/T$.
    \item La mayor parte de la energía se concentra en las bajas frecuencias bajo la envolvente sinc.
    \item Si el período $T$ disminuye, aumenta la separación entre armónicos ($2\pi/T$).
    \item Si la anchura del pulso $\tau$ disminuye, el contenido en frecuencia se extiende sobre un rango mayor.
\end{itemize}
\end{tcolorbox}

\section{Integral y Transformada de Fourier (Señales No Periódicas)}

Cuando el período tiende a infinito ($T \to \infty$), la separación entre armónicos tiende a cero ($\Delta\omega \to 0$) y el espectro discreto se transforma en un espectro continuo.

\begin{equation}
    F(\omega) = \mathcal{F}\{f(t)\} = \int_{-\infty}^{\infty} f(t) e^{-j\omega t} \, dt
\end{equation}
\begin{equation}
    f(t) = \mathcal{F}^{-1}\{F(\omega)\} = \frac{1}{2\pi} \int_{-\infty}^{\infty} F(\omega) e^{j\omega t} \, d\omega
\end{equation}

\subsection{Pulso Rectangular Aislado}
Para un pulso rectangular de amplitud $V$ y duración $\tau$ centrado en $t=0$:
\begin{equation}
    F(\omega) = V \tau \frac{\sin(\omega \tau / 2)}{\omega \tau / 2}
\end{equation}

\section{Ancho de Banda de una Señal}

El ancho de banda es la medida de la dispersión frecuencial donde se concentra la mayor parte de la potencia de la señal.

\begin{itemize}[leftmargin=*]
    \item \textbf{Criterio 1 (Primer corte por cero)}: Frecuencia dada por el primer cero del espectro. Para un pulso de anchura $\tau$:
    \begin{equation}
        \omega_{AB} = \frac{2\pi}{\tau} \text{ rad/s} \quad \iff \quad f_{AB} = \frac{1}{\tau} \text{ Hz}
    \end{equation}
    \item \textbf{Criterio 2 (Banda limitada)}: Frecuencia máxima a partir de la cual el espectro es estrictamente nulo.
    \item \textbf{Criterio 3 (Frecuencias desplazadas)}: Diferencia entre la frecuencia máxima y mínima ($\Delta f = f_{max} - f_{min}$).
\end{itemize}

\section{Sistemas Lineales Estacionarios (LTI) y Respuesta en Frecuencia}

\subsection{Definición de Distorsión y Tipos de Sistemas}
La distorsión es la alteración en la forma de onda de una señal al atravesar un sistema.
\begin{itemize}[leftmargin=*]
    \item \textbf{Sistemas Lineales}: No generan nuevas componentes de frecuencia; solo modifican la amplitud y la fase de las componentes de entrada.
    \item \textbf{Sistemas No Lineales}: Introducen nuevas componentes frecuenciales en la salida.
\end{itemize}

\subsection{Función de Transferencia}
Un sistema LTI se caracteriza unívocamente por su función de transferencia $H(\omega)$, que relaciona la transformada de Fourier de la salida $G(\omega)$ con la de la entrada $F(\omega)$:
\begin{equation}
    H(\omega) = \frac{G(\omega)}{F(\omega)} \implies G(\omega) = H(\omega) F(\omega)
\end{equation}
Ejemplo para una red eléctrica RC serie (salida en el condensador):
\begin{equation}
    H(\omega) = \frac{1}{1 + j\omega RC}
\end{equation}

\subsection{Teorema de la Respuesta Lineal}
\begin{itemize}[leftmargin=*]
    \item \textbf{Entrada senoidal trigonométrica}: Si $f(t) = A \cos(\omega_1 t)$, la salida es:
    \begin{equation}
        g(t) = A |H(\omega_1)| \cos(\omega_1 t + \theta(\omega_1))
    \end{equation}
    \item \textbf{Entrada exponencial compleja}: Si $f(t) = A e^{j \omega_0 t}$, la salida es:
    \begin{equation}
        g(t) = H(\omega_0) A e^{j \omega_0 t} = |H(\omega_0)| A e^{j(\omega_0 t + \theta(\omega_0))}
    \end{equation}
\end{itemize}

\subsection{Ancho de Banda del Sistema}
El ancho de banda $B_s$ es la frecuencia superior que el sistema deja pasar:
\begin{itemize}[leftmargin=*]
    \item \textbf{Criterio de -3 dB}: Frecuencia $B_s$ donde el módulo decae a $|H(2\pi B_s)| = 0.707 = 1/\sqrt{2}$ de su valor máximo (reducción a la mitad de la potencia).
    \item \textbf{Criterio de primer paso por cero}: Frecuencia en la cual el módulo $|H(\omega)|$ se anula por primera vez.
\end{itemize}

\subsection{Estudio Cualitativo de la Respuesta en Frecuencia}
Dada una señal de ancho de banda $B$ que entra a un sistema de ancho de banda $B_s$:
\begin{itemize}[leftmargin=*]
    \item \textbf{Caso $B > B_s$}: La salida está fuertemente distorsionada. La forma final la impone el sistema y su ancho de banda efectivo pasa a ser $B_s$.
    \item \textbf{Caso $B \sim B_s$}: Distorsión moderada; la salida conserva cierta semejanza con la entrada.
    \item \textbf{Caso $B < B_s$}: Transmisión sin distorsión; la salida es una réplica fiel de la entrada.
\end{itemize}

\section{Perturbaciones en la Transmisión de Señales}

\begin{tcolorbox}[cajaResumen, title={Tipos de Perturbaciones}]
\begin{itemize}[leftmargin=*]
    \item \textbf{Atenuación}: Pérdida inevitable de potencia por la distancia. Reduce la amplitud pero no altera la forma de onda. Se mide en dB:
    \begin{equation}
        N = 10 \log_{10}\left(\frac{P_{enviada}}{P_{recibida}}\right) \text{ dB}
    \end{equation}
    \item \textbf{Distorsión de Atenuación}: Producida cuando el medio atenúa de forma desigual a las diferentes frecuencias. Se corrige mediante ecualizadores de atenuación.
    \item \textbf{Distorsión de Retardo (o Fase)}: Ocurre cuando las componentes de distinta frecuencia se propagan a velocidades diferentes, llegando al receptor en instantes distintos. Se corrige mediante ecualizadores de retardo.
    \item \textbf{Ruido}: Procesos aleatorios aditivos.
    \begin{itemize}
        \item \textit{Térmico o Blanco}: Inevitable, origen agitación térmica. Potencia uniforme en todo el espectro.
        \item \textit{Diafonía (Crosstalk)}: Acoplamiento electromagnético indeseado entre líneas o cables próximos.
        \item \textit{Impulsivo}: Picos de corta duración y elevada amplitud (conmutaciones, tormentas). Muy destructivo en transmisión de datos digitales.
    \end{itemize}
\end{itemize}
\end{tcolorbox}

\subsection{Condiciones para un Medio de Transmisión Ideal}
Para lograr una transmisión totalmente libre de distorsión, la función de transferencia del medio debe cumplir:
\begin{equation}
    |H(\omega)| = K \text{ (constante)}, \quad \theta(\omega) = -\omega t_d \text{ (fase lineal con pendiente negativa)}
\end{equation}
Si $K = 1$, el medio es ideal puro (la señal ni se atenúa ni se distorsiona, solo experimenta un retardo $t_d$).

\section{Formulario Completo del Tema 2}

\begin{tcolorbox}[cajaFormulario, title={Formulario General: Señales y Series de Fourier}]
\textbf{1. Parámetros de Señales Senoidales y Ondas}:
\[
f = \frac{1}{T}, \quad \omega = 2\pi f = \frac{2\pi}{T}, \quad \lambda = \frac{c}{f} = c \cdot T
\]

\textbf{2. Serie Compleja de Fourier (Señales Periódicas)}:
\[
f(t) = \frac{1}{T} \sum_{n=-\infty}^{\infty} c_n e^{j n \omega_1 t}, \quad c_n = \int_{-T/2}^{T/2} f(t) e^{-j n \omega_1 t} \, dt
\]

\textbf{3. Relación entre Coeficientes Trigonométricos y Complejos}:
\[
c_n = a_n - j b_n, \quad |c_n| = \sqrt{a_n^2 + b_n^2}, \quad \theta_n = \arctan\left(-\frac{b_n}{a_n}\right)
\]

\textbf{4. Potencia Media (Teorema de Parseval para Señales Periódicas)}:
\[
\bar{P} = \frac{1}{T} \int_{-T/2}^{T/2} |f(t)|^2 \, dt = \sum_{n=-\infty}^{\infty} \left|\frac{c_n}{T}\right|^2
\]

\textbf{5. Tren de Pulsos Rectangulares Periódico}:
\[
c_0 = V\tau, \quad c_n = V\tau \frac{\sin(\omega_n \tau / 2)}{\omega_n \tau / 2}, \quad \omega_{AB} = \frac{2\pi}{\tau} \text{ rad/s}
\]
\end{tcolorbox}

\begin{tcolorbox}[cajaFormulario, title={Formulario General: Transformada de Fourier y Sistemas LTI}]
\textbf{1. Par de Transformadas de Fourier}:
\[
F(\omega) = \int_{-\infty}^{\infty} f(t) e^{-j\omega t} \, dt, \quad f(t) = \frac{1}{2\pi} \int_{-\infty}^{\infty} F(\omega) e^{j\omega t} \, d\omega
\]

\textbf{2. Propiedades de la Transformada de Fourier}:
\begin{itemize}[leftmargin=*]
    \item Linealidad: $\mathcal{F}\{a f_1(t) + b f_2(t)\} = a F_1(\omega) + b F_2(\omega)$
    \item Componente de continua: $F(0) = \int_{-\infty}^{\infty} f(t) \, dt$
    \item Escalado temporal: $\mathcal{F}\{f(at)\} = \frac{1}{|a|} F\left(\frac{\omega}{a}\right)$
    \item Desplazamiento temporal: $\mathcal{F}\{f(t - t_0)\} = F(\omega) e^{-j \omega t_0}$
    \item Modulación frecuencial: $\mathcal{F}\{f(t) e^{j \omega_0 t}\} = F(\omega - \omega_0)$
    \item Modulación cosenoidal: $\mathcal{F}\{f(t) \cos(\omega_0 t)\} = \frac{F(\omega - \omega_0) + F(\omega + \omega_0)}{2}$
    \item Derivación: $\mathcal{F}\left\{\frac{df(t)}{dt}\right\} = j \omega F(\omega)$
    \item Integración: $\mathcal{F}\left\{\int_{-\infty}^{t} f(\tau) d\tau\right\} = \frac{F(\omega)}{j\omega}$
\end{itemize}

\textbf{3. Respuesta de Sistemas LTI}:
\[
G(\omega) = H(\omega) F(\omega), \quad H(\omega) = |H(\omega)| e^{j\theta(\omega)}
\]

\textbf{4. Medidas de Potencia y Atenuación en Decibelios}:
\[
\text{Relación de Potencias (dB)} = 10 \log_{10}\left(\frac{P_A}{P_B}\right)
\]
\[
\text{Atenuación (dB)} = 10 \log_{10}\left(\frac{P_{enviada}}{P_{recibida}}\right)
\]
\end{tcolorbox}

% PROBLEMA 1: Reconstrucción desde Espectros
% ==========================================
\begin{problema}[Reconstrucción de una señal a partir de sus espectros discretos]
Dada una señal periódica $f(t)$ de período $T$, se conocen sus espectros discretos de amplitud y fase. La señal solo contiene componentes no nulas para la frecuencia fundamental $\omega_1 = 5\text{ rad/s}$ y el tercer armónico $\omega_3 = 15\text{ rad/s}$.

Los datos de los coeficientes de Fourier son los siguientes:
\begin{itemize}
    \item Para $\omega_1 = 5\text{ rad/s}$: $\frac{|c_1|}{T} = 0.5$ y fase $\angle c_1 = -\frac{\pi}{4}\text{ rad}$.
    \item Para $\omega_3 = 15\text{ rad/s}$: $\frac{|c_3|}{T} = 0.25$ y fase $\angle c_3 = \frac{\pi}{3}\text{ rad}$.
\end{itemize}

\begin{enumerate}
    \item Obtenga la expresión de la señal $f(t)$ en el dominio del tiempo en su forma trigonométrica compacta.
    \item Obtenga los coeficientes complejos $c_1, c_{-1}, c_3, c_{-3}$ y escriba la señal en su forma exponencial compleja, verificando que coincide con la forma trigonométrica.
\end{enumerate}

\tcblower

\textbf{1. Forma trigonométrica compacta (Método 1):}

La forma trigonométrica compacta del desarrollo en serie de Fourier para una señal periódica viene dada por:
\[ f(t) = \frac{a_0}{T} + \frac{2}{T} \sum_{n=1}^{\infty} |c_n| \cos(\omega_n t + \theta_n) \]

Dado que la señal no presenta componente continua ($a_0 = 0$) y únicamente existen los armónicos $n=1$ ($\omega_1 = 5\text{ rad/s}$) y $n=3$ ($\omega_3 = 15\text{ rad/s}$), sustituimos los valores correspondientes:
\[ f(t) = \frac{2}{T} \left( |c_1| \cos(5t + \angle c_1) + |c_3| \cos(15t + \angle c_3) \right) \]

Sustituyendo los datos $\frac{|c_1|}{T} = 0.5$, $\angle c_1 = -\frac{\pi}{4}$, $\frac{|c_3|}{T} = 0.25$ y $\angle c_3 = \frac{\pi}{3}$:
\[ f(t) = 2 \cdot 0.5 \cos\left(5t - \frac{\pi}{4}\right) + 2 \cdot 0.25 \cos\left(15t + \frac{\pi}{3}\right) \]
\[ f(t) = \cos\left(5t - \frac{\pi}{4}\right) + 0.5 \cos\left(15t + \frac{\pi}{3}\right) \]

\textbf{2. Forma exponencial compleja (Método 2):}

Los coeficientes complejos de Fourier se obtienen a partir de su módulo y su fase ($c_n = |c_n| e^{j \theta_n}$):
\[ c_1 = 0.5 T e^{-j \frac{\pi}{4}}, \quad c_{-1} = 0.5 T e^{j \frac{\pi}{4}} \]
\[ c_3 = 0.25 T e^{j \frac{\pi}{3}}, \quad c_{-3} = 0.25 T e^{-j \frac{\pi}{3}} \]

Sustituyendo en la expresión exponencial de la serie de Fourier $f(t) = \frac{1}{T} \sum_{n=-\infty}^{\infty} c_n e^{j \omega_n t}$:
\[ f(t) = 0.5 e^{-j \frac{\pi}{4}} e^{j 5 t} + 0.5 e^{j \frac{\pi}{4}} e^{-j 5 t} + 0.25 e^{j \frac{\pi}{3}} e^{j 15 t} + 0.25 e^{-j \frac{\pi}{3}} e^{-j 15 t} \]

Agrupando los pares conjugados:
\[ f(t) = 0.5 \left( e^{j \left(5t - \frac{\pi}{4}\right)} + e^{-j \left(5t - \frac{\pi}{4}\right)} \right) + 0.25 \left( e^{j \left(15t + \frac{\pi}{3}\right)} + e^{-j \left(15t + \frac{\pi}{3}\right)} \right) \]

Aplicando la fórmula de Euler $\cos(x) = \frac{e^{jx} + e^{-jx}}{2}$:
\[ f(t) = \cos\left(5t - \frac{\pi}{4}\right) + 0.5 \cos\left(15t + \frac{\pi}{3}\right) \]

Se comprueba que ambos métodos conducen al mismo resultado en el dominio del tiempo.

\end{problema}

% PROBLEMA 2: Propiedades de Serie de Fourier
% ==========================================
\begin{problema}[Serie de Fourier de un par de pulsos mediante propiedades]
Se considera un tren de pulsos rectangulares simétrico respecto al origen $p(t)$, de amplitud $V$, anchura $\tau$ y período $T$. Sus coeficientes de Fourier en forma compleja son conocidos y valen:
\[ c_n^{(p)} = V \tau \frac{\sin(\omega_n \tau / 2)}{\omega_n \tau / 2} \]

A partir de $p(t)$, se construye una nueva señal periódica $f(t) = f_1(t) + f_2(t)$, definida por la superposición de:
\begin{itemize}
    \item $f_1(t) = p(t - \tau/2)$: pulso desplazado a la derecha una cantidad $t_0 = \tau/2$.
    \item $f_2(t) = -p(t + \tau/2)$: pulso de amplitud invertida y desplazado a la izquierda una cantidad $t_0 = \tau/2$.
\end{itemize}

\begin{enumerate}
    \item Aplique el teorema de desplazamiento en el tiempo y la propiedad de linealidad para obtener la expresión de los coeficientes $c_n$ de la serie compleja de Fourier de $f(t)$.
    \item Simplifique los coeficientes utilizando las fórmulas de Euler e indique el tipo de simetría que presenta la señal $f(t)$ en el dominio temporal.
\end{enumerate}

\tcblower

\textbf{Resolución:}

\textbf{1. Aplicación de las propiedades de desplazamiento y linealidad:}

Según el teorema de desplazamiento en el tiempo, si $p(t) \leftrightarrow c_n^{(p)}$, entonces:
\[ f_1(t) = p\left(t - \frac{\tau}{2}\right) \leftrightarrow c_n^{(1)} = c_n^{(p)} e^{-j \omega_n \frac{\tau}{2}} \]
\[ f_2(t) = -p\left(t + \frac{\tau}{2}\right) \leftrightarrow c_n^{(2)} = -c_n^{(p)} e^{+j \omega_n \frac{\tau}{2}} \]

Por la propiedad de linealidad, los coeficientes de $f(t) = f_1(t) + f_2(t)$ son la suma de los coeficientes individuales:
\[ c_n = c_n^{(1)} + c_n^{(2)} = c_n^{(p)} \left( e^{-j \omega_n \frac{\tau}{2}} - e^{+j \omega_n \frac{\tau}{2}} \right) \]

\textbf{2. Simplificación y análisis de simetría:}

Sustituyendo la expresión de $c_n^{(p)}$:
\[ c_n = V \tau \frac{\sin(\omega_n \tau / 2)}{\omega_n \tau / 2} \left( e^{-j \omega_n \frac{\tau}{2}} - e^{+j \omega_n \frac{\tau}{2}} \right) \]

Haciendo uso de la identidad de Euler para la función seno, $\sin(x) = \frac{e^{jx} - e^{-jx}}{2j} \implies e^{-jx} - e^{jx} = -2j \sin(x)$, asignando $x = \frac{\omega_n \tau}{2}$:
\[ e^{-j \omega_n \frac{\tau}{2}} - e^{+j \omega_n \frac{\tau}{2}} = -2j \sin\left(\frac{\omega_n \tau}{2}\right) \]

Sustituyendo este resultado en la expresión de $c_n$:
\[ c_n = V \tau \frac{\sin(\omega_n \tau / 2)}{\omega_n \tau / 2} \left( -2j \sin\left(\frac{\omega_n \tau}{2}\right) \right) \]
\[ c_n = -2j V \tau \frac{\sin^2(\omega_n \tau / 2)}{\omega_n \tau / 2} \]

\textbf{Análisis de simetría:}
Dado que todos los coeficientes $c_n$ son números imaginarios puros y cumplen que $c_{-n} = -c_n$, la señal temporal $f(t)$ es una función \textbf{impar} ($f(-t) = -f(t)$). En consecuencia, su desarrollo en serie trigonométrica de Fourier consta únicamente de términos en senos ($a_n = 0$).

\end{problema}

% ==========================================
% PROBLEMA 1: Transformada analítica y ancho de banda
% ==========================================
\begin{problema}[Transformada de Fourier de un pulso exponencial bilateral e interpretación frecuencial]
Dada la señal no periódica en el dominio del tiempo descrita por una función exponencial bilateral par:
\[ f(t) = e^{-a|t|}, \quad \text{con } a > 0 \]

\begin{enumerate}
    \item Obtenga la expresión analítica de su Transformada de Fourier $F(\omega)$ aplicando directamente la definición de la integral de Fourier.
    \item Compruebe las propiedades de simetría del espectro en frecuencia $F(\omega)$ a partir de la naturaleza de la señal temporal $f(t)$.
    \item Calcule la componente de continua $F(0)$ y verifique la propiedad del área bajo la curva temporal.
    \item Determine el ancho de banda a $-3\text{ dB}$ en frecuencia angular ($\omega_{3\text{dB}}$), definido como la frecuencia en la cual el módulo del espectro decae a $\frac{1}{\sqrt{2}}$ de su valor máximo.
\end{enumerate}

\tcblower

\textbf{1. Obtención de la Transformada de Fourier $F(\omega)$:}

Dado que $|t| = -t$ para $t < 0$ y $|t| = t$ para $t \ge 0$, dividimos la integral de Fourier en dos intervalos:
\[ F(\omega) = \int_{-\infty}^{\infty} e^{-a|t|} e^{-j\omega t} \, dt = \int_{-\infty}^{0} e^{at} e^{-j\omega t} \, dt + \int_{0}^{\infty} e^{-at} e^{-j\omega t} \, dt \]

Agrupando los exponentes:
\[ F(\omega) = \int_{-\infty}^{0} e^{(a - j\omega)t} \, dt + \int_{0}^{\infty} e^{-(a + j\omega)t} \, dt \]

Resolviendo ambas integrales inmediatas:
\[ F(\omega) = \left[ \frac{e^{(a - j\omega)t}}{a - j\omega} \right]_{-\infty}^{0} + \left[ \frac{e^{-(a + j\omega)t}}{-(a + j\omega)} \right]_{0}^{\infty} \]

Evaluando en los límites de integración (sabiendo que $e^{(a - j\omega)(-\infty)} = 0$ y $e^{-(a + j\omega)(\infty)} = 0$ al ser $a > 0$):
\[ F(\omega) = \frac{1}{a - j\omega} - \left( 0 - \frac{1}{a + j\omega} \right) = \frac{1}{a - j\omega} + \frac{1}{a + j\omega} \]

Sumando ambas fracciones mediante el denominador común $(a - j\omega)(a + j\omega) = a^2 + \omega^2$:
\[ F(\omega) = \frac{(a + j\omega) + (a - j\omega)}{a^2 + \omega^2} = \frac{2a}{a^2 + \omega^2} \]

\textbf{2. Análisis de simetría:}

Dado que $f(t) = e^{-a|-t|} = e^{-a|t|} = f(-t)$, la señal temporal es una función \textbf{real y par}.
Consecuentemente, su Transformada de Fourier $F(\omega) = \frac{2a}{a^2 + \omega^2}$ es una función \textbf{real pura y par} en el dominio de la frecuencia, cumpliendo que $F(-\omega) = F(\omega)$ e $\text{Im}\{F(\omega)\} = 0$.

\textbf{3. Componente de continua y propiedad del área:}

La componente de continua se obtiene evaluando $F(\omega)$ en $\omega = 0$:
\[ F(0) = \frac{2a}{a^2 + 0^2} = \frac{2}{a} \]

Por las propiedades de la Transformada de Fourier, $F(0)$ equivale al área bajo la curva de $f(t)$:
\[ \text{Área} = \int_{-\infty}^{\infty} e^{-a|t|} \, dt = 2 \int_{0}^{\infty} e^{-at} \, dt = 2 \left[ \frac{e^{-at}}{-a} \right]_{0}^{\infty} = 2 \left( 0 - \left( -\frac{1}{a} \right) \right) = \frac{2}{a} \]
Se verifica la concordancia exacta entre ambos resultados.

\textbf{4. Cálculo del ancho de banda a $-3\text{ dB}$ ($\omega_{3\text{dB}}$):}

El valor máximo del espectro de amplitud se alcanza en el origen: $F_{\text{máx}} = F(0) = \frac{2}{a}$. La condición a $-3\text{ dB}$ establece que:
\[ F(\omega_{3\text{dB}}) = \frac{1}{\sqrt{2}} F(0) \implies \frac{2a}{a^2 + \omega_{3\text{dB}}^2} = \frac{1}{\sqrt{2}} \left( \frac{2}{a} \right) \]

Simplificando los factores comunes:
\[ \frac{a}{a^2 + \omega_{3\text{dB}}^2} = \frac{1}{\sqrt{2} a} \implies a^2 + \omega_{3\text{dB}}^2 = \sqrt{2} a^2 \]
\[ \omega_{3\text{dB}}^2 = (\sqrt{2} - 1) a^2 \implies \omega_{3\text{dB}} = a \sqrt{\sqrt{2} - 1} \approx 0{,}6436 \, a \text{ rad/s} \]

\end{center}

\end{problema}

% PROBLEMA 2: Descomposición por desplazamiento y muestreo

\begin{problema}[Transformada de Fourier por desplazamiento y coeficientes de la Serie de Fourier]
Se considera la señal no periódica $f(t)$ formada por dos pulsos triangulares opuestos de amplitud $V$ y anchura de base $2\tau$. El primer pulso es positivo y está centrado en $t = -\tau$, mientras que el segundo pulso es negativo y está centrado en $t = +\tau$.

Se conoce que la Transformada de Fourier de un único pulso triangular centrado en el origen de altura $V$ y anchura de base $2\tau$ es:
\[ F_{\text{tri}}(\omega) = V \tau \left( \frac{\sin(\omega \tau / 2)}{\omega \tau / 2} \right)^2 \]

\begin{enumerate}
    \item Exprese $f(t)$ en función del pulso triangular básico $f_{\text{tri}}(t)$ y determine su Transformada de Fourier $F(\omega)$ utilizando las propiedades de linealidad y desplazamiento temporal.
    \item Analice el resultado obtenido y justifique por qué $F(\omega)$ es imaginaria pura e impar.
    \item Asumiendo que esta configuración de pulsos se repite periódicamente con período $T = 4\tau$, determine la expresión analítica de los coeficientes $c_n$ de la Serie de Fourier compleja y halle los valores de $c_0, c_1, c_2, c_3$ y $c_4$.
\end{enumerate}

\tcblower

\textbf{1. Obtención de $F(\omega)$ por propiedades:}

La señal $f(t)$ se puede descomponer como la suma de un pulso triangular positivo adelantado un tiempo $\tau$ y un pulso triangular negativo retrasado un tiempo $\tau$:
\[ f(t) = f_{\text{tri}}(t + \tau) - f_{\text{tri}}(t - \tau) \]

Aplicando la propiedad de desplazamiento temporal $\mathcal{F}\{g(t \pm t_0)\} = G(\omega) e^{\pm j\omega t_0}$ y la linealidad de la Transformada de Fourier:
\[ F(\omega) = F_{\text{tri}}(\omega) e^{j\omega \tau} - F_{\text{tri}}(\omega) e^{-j\omega \tau} = F_{\text{tri}}(\omega) \left( e^{j\omega \tau} - e^{-j\omega \tau} \right) \]

Aplicando la relación de Euler para la función seno, $e^{j\omega \tau} - e^{-j\omega \tau} = 2j \sin(\omega \tau)$:
\[ F(\omega) = 2j \sin(\omega \tau) F_{\text{tri}}(\omega) \]

Sustituyendo la expresión del pulso triangular centrado $F_{\text{tri}}(\omega)$:
\[ F(\omega) = 2j V \tau \sin(\omega \tau) \left( \frac{\sin(\omega \tau / 2)}{\omega \tau / 2} \right)^2 \]

\textbf{2. Análisis de simetría:}

Dado que $f(-t) = f_{\text{tri}}(-t + \tau) - f_{\text{tri}}(-t - \tau) = f_{\text{tri}}(t + \tau) - f_{\text{tri}}(t - \tau) = -f(t)$, la señal es una función \textbf{real e impar} en el dominio del tiempo.
Como propiedad fundamental, la Transformada de Fourier de toda función real e impar es una función \textbf{imaginaria pura e impar} en el dominio de la frecuencia:
\[ F(-\omega) = 2j V \tau \sin(-\omega \tau) \left( \frac{\sin(-\omega \tau / 2)}{-\omega \tau / 2} \right)^2 = -F(\omega) \]

\textbf{3. Extensión periódica y cálculo de los coeficientes $c_n$:}

Para un período $T = 4\tau$, la frecuencia fundamental del desarrollo en serie es:
\[ \omega_1 = \frac{2\pi}{T} = \frac{2\pi}{4\tau} = \frac{\pi}{2\tau} \implies \omega_n = n \omega_1 = \frac{n\pi}{2\tau} \]

Los coeficientes de la serie compleja de Fourier se obtienen muestreando el espectro continuo:
\[ c_n = \frac{1}{T} F(\omega_n) = \frac{1}{4\tau} \left[ 2j V \tau \sin\left(\frac{n\pi}{2}\right) \left( \frac{\sin(n\pi / 4)}{n\pi / 4} \right)^2 \right] \]
\[ c_n = \frac{j V}{2} \sin\left(\frac{n\pi}{2}\right) \left( \frac{\sin(n\pi / 4)}{n\pi / 4} \right)^2 \]

Evaluando para los primeros armónicos ($n = 0, 1, 2, 3, 4$):
\begin{itemize}
    \item \textbf{Para $n = 0$}: $c_0 = \frac{1}{T} F(0) = 0$ (ya que $\sin(0) = 0$). El nivel de continua es nulo porque las áreas del pulso positivo y negativo se cancelan.
    \item \textbf{Para $n = 1$}: $\omega_1 = \frac{\pi}{2\tau}$.
    \[ c_1 = \frac{j V}{2} \sin\left(\frac{\pi}{2}\right) \left( \frac{\sin(\pi / 4)}{\pi / 4} \right)^2 = \frac{j V}{2} (1) \left( \frac{\sqrt{2}/2}{\pi / 4} \right)^2 = \frac{j V}{2} \left( \frac{2\sqrt{2}}{\pi} \right)^2 = j V \frac{4}{\pi^2} \]
    \item \textbf{Para $n = 2$}: $\omega_2 = \frac{\pi}{\tau}$.
    \[ c_2 = \frac{j V}{2} \sin(\pi) \left( \frac{\sin(\pi / 2)}{\pi / 2} \right)^2 = 0 \quad (\text{al ser } \sin(\pi) = 0) \]
    \item \textbf{Para $n = 3$}: $\omega_3 = \frac{3\pi}{2\tau}$.
    \[ c_3 = \frac{j V}{2} \sin\left(\frac{3\pi}{2}\right) \left( \frac{\sin(3\pi / 4)}{3\pi / 4} \right)^2 = \frac{j V}{2} (-1) \left( \frac{\sqrt{2}/2}{3\pi / 4} \right)^2 = -\frac{j V}{2} \left( \frac{2\sqrt{2}}{3\pi} \right)^2 = -j V \frac{4}{9\pi^2} \]
    \item \textbf{Para $n = 4$}: $\omega_4 = \frac{2\pi}{\tau}$.
    \[ c_4 = \frac{j V}{2} \sin(2\pi) \left( \frac{\sin(\pi)}{\pi} \right)^2 = 0 \quad (\text{al ser } \sin(2\pi) = 0) \]
\end{itemize}

\end{problema}
